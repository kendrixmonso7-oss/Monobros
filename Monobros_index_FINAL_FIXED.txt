const MONOBROS_CLOUD_URL = "https://ifppxjauapplmuyqfsal.supabase.co";
const CLOUD_CONFIG_KEY = "monobros_live_supabase_config_v1";
let cloudClient = null;
let cloudSession = null;
let cloudOrgId = null;
let cloudProfile = null;
let cloudBusy = false;

function cloudConfig(){
  try { return JSON.parse(localStorage.getItem(CLOUD_CONFIG_KEY) || "null"); } catch(e){ return null; }
}
function cloudSetMessage(msg, good=false){
  const el=document.getElementById("cloudMessage");
  if(el){ el.textContent=msg; el.style.color=good ? "#16803a" : "#9a3412"; }
}
function cloudSetStatus(text, good=false){
  const bar=document.getElementById("cloudBar"), dot=document.getElementById("cloudStatusDot"), st=document.getElementById("cloudStatusText");
  if(bar) bar.style.display="block";
  if(dot) dot.style.color=good ? "#3ddc84" : "#f5b642";
  if(st) st.textContent=text;
}
function cloudHideGate(){ const g=document.getElementById("cloudGate"); if(g) g.style.display="none"; }
function cloudShowGate(){ const g=document.getElementById("cloudGate"); if(g) g.style.display="block"; }
function saveCloudConfig(){
  const url=(document.getElementById("cloudUrl")?.value || MONOBROS_CLOUD_URL).trim().replace(/\/rest\/v1\/?$/,"");
  const key=(document.getElementById("cloudKey")?.value || "").trim();
  if(!url || !key){ cloudSetMessage("Enter the Project URL and the Publishable key."); return; }
  if(!/^https:\/\/.+\.supabase\.co$/i.test(url) && !/^https:\/\/.+$/i.test(url)){ cloudSetMessage("Use the Supabase Project URL beginning with https://"); return; }
  localStorage.setItem(CLOUD_CONFIG_KEY, JSON.stringify({url, key}));
  cloudSetMessage("Cloud configuration saved on this device. Testing connection...", true);
  cloudInit(true);
}
async function cloudInit(showGateIfNeeded=false){
  const cfg=cloudConfig();
  const urlInput=document.getElementById("cloudUrl");
  if(urlInput && !urlInput.value) urlInput.value=MONOBROS_CLOUD_URL;
  if(!cfg){
    cloudSetStatus("Cloud not configured", false);
    if(showGateIfNeeded) cloudShowGate();
    return false;
  }
  try{
    cloudClient = window.supabase.createClient(cfg.url.replace(/\/$/,""), cfg.key);
    const {data,error}=await cloudClient.auth.getSession();
    if(error) throw error;
    cloudSession=data.session;
    if(!cloudSession){
      cloudSetStatus("Cloud ready — login required", false);
      cloudShowGate();
      return false;
    }
    await cloudLoadProfile();
    cloudHideGate();
    cloudSetStatus("Cloud connected", true);
    return true;
  }catch(e){
    cloudSetStatus("Cloud connection error", false);
    if(showGateIfNeeded) cloudSetMessage("Connection failed: " + (e.message || e));
    cloudShowGate();
    return false;
  }
}
async function cloudCheck(){
  const cfg=cloudConfig();
  if(!cfg){ cloudSetMessage("Save the Project URL and Publishable key first."); return; }
  try{
    const c=window.supabase.createClient(cfg.url.replace(/\/$/,""), cfg.key);
    const {data,error}=await c.auth.getSession();
    if(error) throw error;
    cloudSetMessage("Supabase responded successfully. " + (data.session ? "A login session is active." : "Now enter your admin login."), true);
  }catch(e){ cloudSetMessage("Connection failed: " + (e.message || e)); }
}
async function cloudLogin(){
  const cfg=cloudConfig();
  if(!cfg){ cloudSetMessage("Save your Project URL and Publishable key first."); return; }
  const email=(document.getElementById("loginEmail")?.value || "").trim();
  const password=document.getElementById("loginPassword")?.value || "";
  if(!email || !password){ cloudSetMessage("Enter your admin email and password."); return; }
  try{
    cloudSetMessage("Signing in...");
    cloudClient=window.supabase.createClient(cfg.url.replace(/\/$/,""), cfg.key);
    const {data,error}=await cloudClient.auth.signInWithPassword({email,password});
    if(error) throw error;
    cloudSession=data.session;
    await cloudLoadProfile();
    cloudHideGate();
    cloudSetStatus("Cloud connected", true);
    await cloudSyncNow(true);
  }catch(e){ cloudSetMessage("Login failed: " + (e.message || e)); }
}
async function cloudLogout(){
  try{ if(cloudClient) await cloudClient.auth.signOut(); }catch(e){}
  cloudSession=null; cloudOrgId=null; cloudProfile=null;
  cloudSetStatus("Logged out", false);
  cloudShowGate();
}
async function cloudLoadProfile(){
  if(!cloudClient || !cloudSession) throw new Error("No authenticated session.");
  const {data,error}=await cloudClient.from("profiles").select("id,organization_id,full_name,role,active").eq("id",cloudSession.user.id).single();
  if(error) throw new Error("Profile not available. Make sure your Monobros ADMIN profile exists and the RLS policies have been installed.");
  if(!data.active) throw new Error("This account is inactive.");
  cloudProfile=data; cloudOrgId=data.organization_id;
  cloudSetStatus("Cloud connected — " + data.role, true);
}
async function cloudSyncNow(silent=false){
  if(cloudBusy) return;
  if(!cloudClient || !cloudSession || !cloudOrgId){
    if(!silent) alert("Log in to the cloud first.");
    return;
  }
  cloudBusy=true; cloudSetStatus("Synchronizing...", true);
  try{
    await cloudPushLocal();
    await cloudPullToLocal();
    if(typeof render==="function") render();
    cloudSetStatus("Cloud synchronized", true);
    if(!silent) alert("Monobros data synchronized successfully.");
  }catch(e){
    console.error(e);
    cloudSetStatus("Sync error", false);
    if(!silent) alert("Cloud sync error: " + (e.message || e));
  }finally{ cloudBusy=false; }
}
function localProducts(){ try{return JSON.parse(localStorage.ppProducts||"[]")}catch(e){return []} }
function localCustomers(){ try{return JSON.parse(localStorage.ppCustomers||"[]")}catch(e){return []} }
function localInvoices(){ try{return JSON.parse(localStorage.ppInvoices||"[]")}catch(e){return []} }
function localPurchases(){ try{return JSON.parse(localStorage.ppPurchases||"[]")}catch(e){return []} }
function localExpenses(){ try{return JSON.parse(localStorage.ppExpenses||"[]")}catch(e){return []} }

async function ensureCloudCustomer(c){
  if(c.cloud_id) return c.cloud_id;
  let q=cloudClient.from("customers").select("id").eq("organization_id",cloudOrgId).eq("name",c.name).limit(1);
  if(c.phone) q=q.eq("phone",c.phone);
  const {data,error}=await q;
  if(error) throw error;
  if(data && data[0]){ c.cloud_id=data[0].id; return c.cloud_id; }
  const {data:ins,error:ie}=await cloudClient.from("customers").insert({
    organization_id:cloudOrgId,name:c.name,phone:c.phone||null,address:c.address||null,opening_balance:0
  }).select("id").single();
  if(ie) throw ie;
  c.cloud_id=ins.id; return ins.id;
}
async function ensureCloudProduct(p){
  if(p.cloud_id) return p.cloud_id;
  const {data,error}=await cloudClient.from("products").select("id").eq("organization_id",cloudOrgId).eq("name",p.name).eq("unit",p.size||"unit").limit(1);
  if(error) throw error;
  if(data && data[0]){ p.cloud_id=data[0].id; return p.cloud_id; }
  const {data:ins,error:ie}=await cloudClient.from("products").insert({
    organization_id:cloudOrgId,name:p.name,unit:p.size||"unit",sale_price:Number(p.price||0),cost_price:Number(p.cost||0),stock:Number(p.stock||0),reorder_level:5,active:true
  }).select("id").single();
  if(ie) throw ie;
  p.cloud_id=ins.id; return ins.id;
}
async function cloudPushLocal(){
  const ps=localProducts();
  for(const p of ps){
    const id=await ensureCloudProduct(p);
    const {error}=await cloudClient.from("products").update({
      name:p.name,unit:p.size||"unit",sale_price:Number(p.price||0),cost_price:Number(p.cost||0),stock:Number(p.stock||0),active:true,updated_at:new Date().toISOString()
    }).eq("id",id).eq("organization_id",cloudOrgId);
    if(error) throw error;
  }
  localStorage.ppProducts=JSON.stringify(ps);

  const cs=localCustomers();
  for(const c of cs) await ensureCloudCustomer(c);
  localStorage.ppCustomers=JSON.stringify(cs);

  const invs=localInvoices();
  for(const inv of invs){
    if(inv.cloud_id) continue;
    let {data:existing,error:qe}=await cloudClient.from("invoices").select("id").eq("organization_id",cloudOrgId).eq("invoice_no",inv.no).limit(1);
    if(qe) throw qe;
    if(existing && existing[0]){ inv.cloud_id=existing[0].id; continue; }
    const customer=cs.find(x=>x.name===inv.customer);
    const customerId=customer ? await ensureCloudCustomer(customer) : null;
    const amountPaid=inv.paymentStatus==="Paid" ? Number(inv.total||0) : inv.paymentStatus==="Part Paid" ? 0 : 0;
    const {data:ins,error:ie}=await cloudClient.from("invoices").insert({
      organization_id:cloudOrgId,invoice_no:inv.no,customer_id:customerId,status:"POSTED",
      payment_status:inv.paymentStatus||"Paid",payment_method:inv.paymentMethod||"Cash",
      subtotal:Number(inv.total||0),discount:0,total:Number(inv.total||0),amount_paid:amountPaid,
      balance:Math.max(0,Number(inv.total||0)-amountPaid),created_by:cloudSession.user.id,created_at:inv.date||new Date().toISOString()
    }).select("id").single();
    if(ie) throw ie;
    inv.cloud_id=ins.id;
    const items=[];
    for(const item of (inv.items||[])){
      const p=ps.find(x=>x.name===item.name && (x.size||"unit")===(item.size||"unit"));
      if(p){
        const pid=await ensureCloudProduct(p);
        items.push({invoice_id:ins.id,product_id:pid,quantity:Number(item.qty||0),unit_price:Number(item.price||0),line_total:Number(item.qty||0)*Number(item.price||0)});
      }
    }
    if(items.length){
      const {error:ii}=await cloudClient.from("invoice_items").insert(items);
      if(ii) throw ii;
    }
  }
  localStorage.ppInvoices=JSON.stringify(invs);

  const exps=localExpenses();
  for(const e of exps){
    if(e.cloud_id) continue;
    const {data:ins,error}=await cloudClient.from("expenses").insert({
      organization_id:cloudOrgId,category:e.category||null,description:e.description,amount:Number(e.amount||0),created_by:cloudSession.user.id,created_at:e.date||new Date().toISOString()
    }).select("id").single();
    if(error) throw error;
    e.cloud_id=ins.id;
  }
  localStorage.ppExpenses=JSON.stringify(exps);
}
async function cloudPullToLocal(){
  const {data:ps,error:pe}=await cloudClient.from("products").select("*").eq("organization_id",cloudOrgId).eq("active",true).order("created_at");
  if(pe) throw pe;
  if(ps && ps.length){
    const mapped=ps.map((p,i)=>({id:i+1,cloud_id:p.id,name:p.name,size:p.unit,price:Number(p.sale_price||0),cost:Number(p.cost_price||0),stock:Number(p.stock||0)}));
    localStorage.ppProducts=JSON.stringify(mapped);
  }
  const {data:cs,error:ce}=await cloudClient.from("customers").select("*").eq("organization_id",cloudOrgId).order("created_at");
  if(ce) throw ce;
  if(cs){
    localStorage.ppCustomers=JSON.stringify(cs.map((c,i)=>({id:i,name:c.name,phone:c.phone||"",tpin:"",address:c.address||"",cloud_id:c.id})));
  }
  const {data:invs,error:ie}=await cloudClient.from("invoices").select("*, invoice_items(*)").eq("organization_id",cloudOrgId).order("created_at",{ascending:false});
  if(ie) throw ie;
  if(invs){
    const products=JSON.parse(localStorage.ppProducts||"[]");
    const customers=JSON.parse(localStorage.ppCustomers||"[]");
    const mapped=invs.map(i=>({
      no:i.invoice_no,date:i.created_at,customer:(customers.find(c=>c.cloud_id===i.customer_id)||{}).name||"Walk-in customer",
      items:(i.invoice_items||[]).map(it=>{const p=products.find(x=>x.cloud_id===it.product_id)||{};return {name:p.name||"Product",size:p.size||"",qty:Number(it.quantity||0),price:Number(it.unit_price||0),cost:Number(p.cost||0)}}),
      total:Number(i.total||0),paymentStatus:i.payment_status||"Paid",paymentMethod:i.payment_method||"Cash",cloud_id:i.id
    }));
    localStorage.ppInvoices=JSON.stringify(mapped);
  }
  const {data:ex,error:ee}=await cloudClient.from("expenses").select("*").eq("organization_id",cloudOrgId).order("created_at",{ascending:false});
  if(ee) throw ee;
  if(ex) localStorage.ppExpenses=JSON.stringify(ex.map(e=>({date:e.created_at,description:e.description,category:e.category||"",amount:Number(e.amount||0),cloud_id:e.id})));
}

  }catch(e){console.warn("Audit log failed",e);}
}
